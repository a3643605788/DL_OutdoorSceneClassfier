outdoor-scene-classifier/
  README.md
  .gitignore
  notebooks/
    dataset_info.ipynb.ipynb ------------------# 下載Intel Scene Classification dataset，並用notebook 顯⽰ 10 張圖⽚
    day5_dataloader_debug.ipynb ---------------# DataLoader 測試與 Debug
  src/ ---------------------------------------------# 放可重用、可被訓練/推論流程 import 的程式碼（模型、資料處理、訓練、評估）
    __init__.py
    data_preprocessing.py ---------------------# 資料前處理程式(完整)
    model.py ----------------------------------# (CNN baseline)
    dataset.py  -------------------------------# 資料前處理程式(一部分)(供evaluate.py使用)
    evaluate.py -------------------------------# 驗證模型在未見過數據（Test Set）上的最終實力
    gradcam.py --------------------------------# 抓出誤判的圖片畫出熱力圖
  tools/ -------------------------------------------# 放一次性或輔助用途的工具腳本（印 summary、檢查資料夾結構、算類別數、測試 dataloader 等）
    print_model_summary.py --------------------# 用 summary() 印出模型結構
    day7_smoke_forward_backward.py ------------# 用來驗證 「資料 → 模型 → loss → 反向傳遞 → 參數更新」 這條路能跑通
  api/
    __init__.py
    main.py -----------------------------------# 定義FastAPI路由，掛載並使用ResNet18模型，在http://127.0.0.1:8000/docs
  data/
    raw/ --------------------------------------# 原始圖片或標註
    processed/ --------------------------------# 前處理後的資料
  outputs/ -----------------------------------------# 將「不可變的程式碼 (src/)」與「變動的實驗結果 (outputs/)」分離。日後調整參數可以拿來對比
    best_baseline.pth -------------------------# 儲存了模型中數百萬個參數，在驗證損失（Val Loss）最低點時的數值
    learning_curve.png ------------------------# 將數據轉化為折線圖，呈現 Loss（誤差）如何下降以及 Accuracy（準確率）如何上升
    metrics.json ------------------------------# 以 JSON 格式完整保留了每一輪（Epoch）的訓練與驗證數值
    confusion_matrix.png ----------------------# 展示了測試集中每一張圖片的「真實類別（Y軸）」與「模型預測類別（X軸）」的對應關係
    misclassified_samples.png -----------------# 模型「看錯」的那些原始照片印出來
  tests/ -------------------------------------------# 測試
    test_api.py -------------------------------# 自動化單元測試：驗證 API 接口穩定性與推論正確性
  .gitignore
  .dockerignore -------------------------------# 防止 Docker 把巨大的 venv、__pycache__ 或訓練用的原始資料也塞進鏡像裡，優化Docker 的 image體積
  Dockerfile ----------------------------------# 確保最終產品裡只有執行檔沒有垃圾
  run_diagnosis.py ----------------------------# 執行畫熱力圖腳本
  train.py ------------------------------------# 把 src/ 裡的零件組裝起來並啟動訓練任務
  requirements.txt ----------------------------# 優化 Docker 鏡像（Image）體積
  README.md -----------------------------------# 紀錄實驗設定
